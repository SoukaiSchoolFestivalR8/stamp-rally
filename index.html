<body>

  <h2>文化祭スタンプラリー</h2>

 

  <!-- カメラ起動ボタン -->

  <button id="start-scan-btn" onclick="startScanner()">📷 カメラでQRを読み込む</button>

  <div id="reader" style="display:none;"></div>

  <!-- スタンプ獲得状況 -->

  <h3>獲得状況</h3>

  <div class="stamp-grid">

    <div id="stamp-booth1" class="stamp-box">ブース 1</div>

    <div id="stamp-booth2" class="stamp-box">ブース 2</div>

    <div id="stamp-booth3" class="stamp-box">ブース 3</div>

  </div>

  <script>

    // ブースごとのIDキー定義（QRコードに仕込む文字列）

    const STAMPS = ["booth1", "booth2", "booth3"];

    let html5QrcodeScanner = null;

    // 画面読み込み時に保存済みスタンプを復元

    window.onload = function() {

      updateUI();

    };

    function startScanner() {

      document.getElementById('reader').style.display = 'block';

      document.getElementById('start-scan-btn').style.display = 'none';

      html5QrcodeScanner = new Html5Qrcode("reader");

      html5QrcodeScanner.start(

        { facingMode: "environment" }, // 背面カメラを優先設定

        { fps: 10, qrbox: { width: 250, height: 250 } },

        onScanSuccess,

        onScanError

      ).catch(err => {

        alert("カメラの起動に失敗しました。カメラ権限を許可してください。");

        stopScanner();

      });

    }

    function onScanSuccess(decodedText) {

      // 読み取ったQRコードテキストが定義したスタンプキーと一致するか判定

      if (STAMPS.includes(decodedText)) {

        // localStorageに保存

        localStorage.setItem("stamp_" + decodedText, "true");

        alert("スタンプを獲得しました！: " + decodedText);

        updateUI();

        stopScanner();

      } else {

        alert("無効なQRコードです");

      }

    }

    function onScanError(errorMessage) {

      // スキャン実行中の定期エラーは無視

    }

    function stopScanner() {

      if (html5QrcodeScanner) {

        html5QrcodeScanner.stop().then(() => {

          document.getElementById('reader').style.display = 'none';

          document.getElementById('start-scan-btn').style.display = 'inline-block';

        }).catch(err => console.error(err));

      }

    }

    // 保存状況に合わせて表示（済マーク等）を更新

    function updateUI() {

      STAMPS.forEach(boothId => {

        const isAcquired = localStorage.getItem("stamp_" + boothId) === "true";

        const element = document.getElementById("stamp-" + boothId);

        if (element) {

          if (isAcquired) {

            element.classList.add("acquired");

          } else {

            element.classList.remove("acquired");

          }

        }

      });

    }

  </script>

</body>
